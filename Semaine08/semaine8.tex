\begin{exercise}
    Trouver les sous-corps de $\Q$.
\end{exercise} 

\begin{exercise}
    Soit $A$ un anneau. $A$ est dit \textit{régulier} ssi $\forall a \in A, \exists u \in A, a u a = a$.
    \begin{questions}
        \item Un corps est-il régulier~?
        \item $\Z$ est-il régulier~?
        \item Donner une condition nécessaire et suffisante sur $n \in \N$ pour que $\Z / n \Z$ soit régulier.
    \end{questions}
\end{exercise}

\begin{exercise}
Soit $A$ un anneau commutatif.
Montrer que $A$ est un corps ssi $\set{0},A$ sont les seuls idéaux de $A$.
\end{exercise}

\begin{exercise}
Soit $a\in\Q^+$ tel que $\sqrt{a}\notin\Q$.
On note $\Q(\sqrt{a})=\set{r+s\sqrt{a}, r,s\in\Q}$.
\begin{questions}
\item Montrer que $\Q(\sqrt{a})$ est un sous-corps de $\R$.
\item $\Q(\sqrt{2})$ et $\Q(\sqrt{3})$ sont-ils isomorphes ?
\end{questions}
\end{exercise}

\begin{exercise}
Soit $G$ un groupe.
On note $Z(G)=\set{z\in G, \forall g\in G, zg=gz}$ le centre de $G$.
\begin{questions}
\item Montrer que $Z(G)$ est un sous-groupe distingué de $G$.
\item Montrer que si $G/Z(G)$ est cyclique, alors $G$ est abélien.
\end{questions}
\end{exercise}

\begin{exercise}
Soit $G$ un groupe de cardinal $p^2$ avec $p$ premier.
Montrer que $G\cong \Z/p^2\Z$ ou $G\cong (\Z/p\Z)^2$.
\end{exercise}

\begin{exercise}
Soit $K$ un corps de cardinal $p\in\mathbb{P}$
\begin{questions}
\item Montrer que $K$ est isomorphe à $\Z/p\Z$.
\item Montrer qu'il existe un unique sous-corps de $K$ de caractéristique $p$ isomorphe à $\Z/p\Z$.
\end{questions}
\end{exercise}
