\begin{exercise}
    Soit $k \in \N$ et $ s_1, s_2, \dots, s_n$ des entiers.
    Montrer que:
    \[
        \binom{\sum_{i=1}^n s_i}{k} = \sum_{\substack{0 \le x_1,x_2,\dots,x_n \le k \\ x_1 + x_2 + \cdots + x_n = k}} \prod_{i=1}^n \binom{s_i}{x_i}
    \]
\end{exercise}

\begin{exercise}

\end{exercise}

\begin{exercise}
Soient $z_1,\ldots,z_k\in\mathbb{C}$ tels que
\[
|z_i|\leq 1
\qquad \text{pour tout } i\in\{1,\ldots,k\}.
\]

Montrer qu'il existe un choix de
\[
\varepsilon_1,\ldots,\varepsilon_k\in\{-1,1\}
\]
tel que
\[
\left|\sum_{i=1}^k \varepsilon_i z_i\right|\leq \sqrt{3}.
\]
\end{exercise}

\begin{exercise}
Soient $z_1,\ldots,z_k\in\mathbb{C}$.
Montrer qu'il existe un choix de
\[
\varepsilon_1,\ldots,\varepsilon_k\in\{-1,1\}
\]
tel que
\[
\left|\sum_{i=1}^k \varepsilon_i z_i\right|
\geq
\frac{2}{\pi}\sum_{i=1}^k |z_i|.
\]
\end{exercise}

\begin{exercise}
Soit $E$ un ensemble. Montrer que les deux assertions suivantes sont équivalentes :
\begin{questions}
    \item $E$ est infini ;
    \item Pour toute application
    \[
    f:E\to E,
    \]
    il existe une partie $A\subset E$, non triviale, c'est-à-dire
    \[
    A\neq\varnothing
    \qquad\text{et}\qquad
    A\neq E,
    \]
    qui est stable par $f$, autrement dit telle que
    \[
    f(A)\subset A.
    \]
\end{questions}
\end{exercise}

\end{questions}
\end{exercise}
