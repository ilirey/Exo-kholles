\begin{exercise}
\begin{questions}
\item Soit $n\in \N^*$. Résoudre l'équation $(1+ix)^n=(1-ix)^n$ d'inconnue $x\in \R$.
\item En déduire la valeur de $\displaystyle\sum_{k=0}^{n-1} \tan\left(\frac{k\pi}{n}\right)$ si $n$ est impair.
\end{questions}
\end{exercise}

\begin{exercise}
    Soit $n \in \N^*$. Calculer~: $\displaystyle\sum_{k=0}^{n-1} \abs {\omega_n^k - 1}$
\end{exercise}

\begin{exercise}
    Soit $k \in \N$ et $ s_1, s_2, \dots, s_n$ des entiers.
    Montrer que:
    \[
        \binom{\sum_{i=1}^n s_i}{k} = \sum_{\substack{0 \le x_1,x_2,\dots,x_n \le k \\ x_1 + x_2 + \cdots + x_n = k}} \prod_{i=1}^n \binom{s_i}{x_i}
    \]
\end{exercise}

\begin{exercise}
    Soient $A(a)$ et $B(b)$ deux points du cercle unité et $P(z)$ un point du plan. On note $H(h)$ le projeté orthogonal de $P$ sur $(AB)$.
    
    Montrer que~: $h = \dfrac{1}{2} (a + b + z - a b \bar{z})$
\end{exercise}
