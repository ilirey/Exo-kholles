\begin{exercise}
Soit $I$ un intervalle, $f$ continue sur $I$ à valeurs dans $[a,b]$ avec $a\lt b\in\R$.\\
Montrer que $\displaystyle\int_If^2\le-ab|I|$.
\end{exercise}

\begin{exercise}
Soit $f$ une fonction continue sur $[0,1]$.\\
Calculer $\lim\limits_{n\to+\infty}\int_0^1f(t^n)dt$
\end{exercise}

\begin{exercise}
Montrer que pour tout $t\gt0, \arctan(t)\ge\frac{t}{1+t^2}$
\end{exercise}

\begin{exercise}
Soit $z_1,\ldots, z_n\in\C$. Montrer qu'il existe $I\subset\intset{1,n}$ tel que $\abs{\sum_{i\in I}z_i}\ge\frac{1}{6}\sum_{i=1}^n\abs{z_i}$
\end{exercise}

\begin{exercise}
Soit $f\in C^0(\R^+,\R^+)$ telle qu'il existe $k\ge0$ tel que pour tout $x\gt0, f(x)\le k\int_0^xf(t)dt$.
Montrer que $f$ est nulle.
\end{exercise}

\begin{exercise}
Soit $f$ une fonction continue sur $[a,b]$ et $n\in\N$ telle que $\forall k\in\intset{0,n},\int_a^bt^kf(t)dt=0$.
Montrer que $f$ s'annule au moins $n+1$ fois sur $[a,b]$.
\end{exercise}

\begin{exercise}
Soit $n\in\N$, on pose $P_n:x\mapsto\sum_{k=0}^n\frac{x^k}{k!}$.\\
Calculer une primitive de $x\mapsto\frac{x^n}{P_n(x)}$
\end{exercise}

\begin{exercise}
\begin{questions}
\item Soit $P\in\C[X]$. Montrer que $\int_{-1}^1P(t)dt=-i\int_0^{\pi}P(e^{ix})e^{ix}dx$.
\item En déduire que pour tout réels $a_1,\ldots,a_n$, on a
$$\sum_{0\le j,k\le n}\frac{a_ja_k}{j+k+1}\le\pi\sum_{k=0}^na_k^2$$
\end{questions}
\end{exercise}

\begin{exercise}
Trouver toutes les fonctions continues $f:\R\to\R$ telles que
pour tout $a\lt b, \frac{1}{b-a}\int_a^bf(t)dt=f\left(\frac{a+b}{2}\right)$
\end{exercise}

\begin{exercise}
Soit a,b réels. Calculer
\[\int_a^b\sqrt{1+\sqrt x}\,dx\]
\end{exercise}

\begin{exercise}
Déterminer une primitive de $$\sin(\ln(x))$$
\end{exercise}
\begin{exercise}
Calculer $$\int_0^{\pi}\frac{x\sin(x)}{1+\cos(x)^2}dx$$
\end{exercise}
\begin{exercise}
Calculer $$\int_0^1\frac{\arctan(x)}{x+1}dx$$
\end{exercise}
\begin{exercise}
Montrer l'existence puis déterminer $$\lim\limits_{x\rightarrow 1^-}\int_x^{x^2}\frac1{\ln(t)}dt$$
\end{exercise}
\begin{exercise}
Soit $f : [a,b]\rightarrow \mathbb R$, $C^0$, tel que $f(a)=0$ et $\forall x\in[a,b], f'(x)\in[0,1]$.
Montrer que $$\int_a^b f^3(x)dx\le(\int_a^b f(x)dx)^2$$
\end{exercise}
