\begin{exercise}
Montrer qu'il n'existe pas de fonction $f:\R\to \R$ continue telle que
$f\circ f=-\text{Id}$.
\end{exercise}

\begin{exercise}
Soit $f:x\mapsto \frac{1}{x}e^{-\frac{1}{\abs{x}}}$ définie sur $\R^*$.
Montrer que $f$ est prolongeable par continuité en $0$ et que la fonction ainsi prolongée est dérivable en $0$.
\end{exercise}

\begin{exercise}
Trouver toutes les fonctions $f:[0,1]\to \R$ continues telles que 
$$\int_0^1 f(t)dt=\frac{1}{3}+\int_0^1(f(t^2))^2dt$$
\end{exercise}
